S := 'int' 'main' '(' ')' '{' BLOCK '}' |;

BLOCK := DECLARATION BLOCK
	| ASSIGNMENT BLOCK
	| CONST_ASSIGNMENT BLOCK
	| SWITCH BLOCK
	| 'eps' |;
LOOP_BLOCK := DECLARATION LOOP_BLOCK
	| ASSIGNMENT LOOP_BLOCK
	| CONST_ASSIGNMENT LOOP_BLOCK
	| SWITCH LOOP_BLOCK
	| 'break' ';' LOOP_BLOCK
	| 'eps'|;

DECLARATION := 'int' id OPTIONAL_INIT ';'|;
OPTIONAL_INIT := '=' EXPRESSION | 'eps'|;
ASSIGNMENT := id '=' EXPRESSION ';'|;
CONST_ASSIGNMENT := 'const' 'int' const_id '=' EXPRESSION ';'|;

EXPRESSION := TERM EXPRESSION_TAIL|;
EXPRESSION_TAIL := operator TERM EXPRESSION_TAIL | 'eps'|;
TERM := '(' EXPRESSION ')' | OPERAND|;
OPERAND := id | CONST_OPERAND|;

CONST_EXPRESSION := CONST_TERM CONST_EXPRESSION_TAIL|;
CONST_EXPRESSION_TAIL := operator CONST_TERM CONST_EXPRESSION_TAIL | 'eps'|;
CONST_TERM := '(' CONST_EXPRESSION ')' | CONST_OPERAND|;
CONST_OPERAND := int_literal | const_id|;

SWITCH := 'switch' '(' EXPRESSION ')' '{' SWITCH_BODY '}'|;
SWITCH_BODY := CASES DEFAULT|;
CASES := CASE CASES | 'eps'|;
CASE := 'case' CONST_EXPRESSION ':' LOOP_BLOCK|;
DEFAULT := 'default' ':' LOOP_BLOCK | 'eps'|;
